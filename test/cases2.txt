public class Main {
    static class InsufficientFundsException extends RuntimeException {
        private final double missing;
        InsufficientFundsException(double missing) { super("Lipsesc " + missing + " lei"); this.missing = missing; }
        double getMissing() { return missing; }
    }
    static void withdraw(double balance, double amount) {
        if (amount > balance) throw new InsufficientFundsException(amount - balance);
    }
    public static void main(String[] args) {
        try { withdraw(50, 80); }
        catch (InsufficientFundsException e) { System.out.println(e.getMessage() + " / " + e.getMissing()); }
        try {
            try { Integer.parseInt("x"); }
            catch (NumberFormatException e) { throw new IllegalStateException("config invalid", e); }
        } catch (IllegalStateException e) {
            System.out.println(e.getMessage() + " <- " + e.getCause().getClass().getName());
        }
        withdraw(1, 2);
    }
}

import java.util.*;
import java.util.stream.*;
public class Main {
    static class Student {
        private final String name; private final String group; private final List<Integer> grades;
        Student(String name, String group, Integer... grades) { this.name = name; this.group = group; this.grades = Arrays.asList(grades); }
        String getName() { return name; } String getGroup() { return group; }
        double average() { return grades.stream().mapToInt(Integer::intValue).average().orElse(0); }
    }
    public static void main(String[] args) {
        List<Student> list = List.of(new Student("Ana", "4LF341", 9, 10, 8), new Student("Bogdan", "4LF342", 6, 7),
                                     new Student("Carmen", "4LF341", 10, 10), new Student("Dan", "4LF342", 5, 4, 6));
        Map<String, Double> avgByGroup = list.stream().collect(Collectors.groupingBy(Student::getGroup, TreeMap::new, Collectors.averagingDouble(Student::average)));
        avgByGroup.forEach((g, a) -> System.out.printf("%s: %.2f%n", g, a));
        String top = list.stream().sorted(Comparator.comparingDouble(Student::average).reversed()).limit(2).map(Student::getName).collect(Collectors.joining(" & "));
        System.out.println(top);
        System.out.println(list.stream().filter(s -> s.average() < 5).map(Student::getName).findFirst().orElse("nimeni"));
        IntSummaryStatistics st = list.stream().flatMap(s -> s.grades.stream()).mapToInt(x -> x).summaryStatistics();
        System.out.println(st.getMin() + " " + st.getMax() + " " + st.getCount() + " " + st.getAverage());
        System.out.println(list.stream().collect(Collectors.groupingBy(Student::getGroup, Collectors.mapping(Student::getName, Collectors.toList()))));
        List<List<Integer>> nested = List.of(List.of(1, 2), List.of(3), List.of());
        System.out.println(nested + " " + nested.stream().mapToInt(List::size).sum());
    }
}

import java.util.*;
interface Payable { double pay(); default String label() { return getClass().getSimpleName() + ": " + pay(); } }
abstract class Employee implements Payable {
    protected final String name;
    Employee(String name) { this.name = name; }
}
class Salaried extends Employee {
    private final double salary;
    Salaried(String n, double s) { super(n); salary = s; }
    public double pay() { return salary; }
}
class Hourly extends Employee {
    private final double rate; private final int hours;
    Hourly(String n, double rate, int hours) { super(n); this.rate = rate; this.hours = hours; }
    public double pay() { return hours > 40 ? 40 * rate + (hours - 40) * rate * 1.5 : hours * rate; }
}
public class Main {
    public static void main(String[] args) {
        List<Payable> staff = new ArrayList<>();
        staff.add(new Salaried("Ana", 5000));
        staff.add(new Hourly("Dan", 30, 45));
        staff.add(() -> 100.0);
        double total = 0;
        for (Payable p : staff) total += p.pay();
        System.out.println(total);
        System.out.println(staff.get(0).label());
        System.out.println(staff.get(1).label());
    }
}

import java.util.*;
import java.util.function.*;
public class Main {
    @FunctionalInterface
    interface Validator<T> { boolean validate(T t); }
    static <T> List<T> keep(List<T> xs, Validator<T> v) {
        List<T> out = new ArrayList<>();
        for (T x : xs) if (v.validate(x)) out.add(x);
        return out;
    }
    public static void main(String[] args) {
        System.out.println(keep(List.of(1, 5, 12, 7, 20), x -> x > 6));
        System.out.println(keep(List.of("ana", "", "ion"), s -> !s.isEmpty()));
        BinaryOperator<Integer> add = Integer::sum;
        Function<String, Integer> len = String::length;
        Supplier<Double> pi = () -> Math.PI;
        System.out.println(add.apply(2, 3) + len.apply("Java") + " " + pi.get());
        Map<String, Function<Double, Double>> ops = new LinkedHashMap<>();
        ops.put("dublu", x -> x * 2); ops.put("patrat", x -> x * x);
        ops.forEach((k, f) -> System.out.println(k + " " + f.apply(3.0)));
    }
}
