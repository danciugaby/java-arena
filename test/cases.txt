public class Main {
    public static void main(String[] args) {
        int sum = 0;
        for (int i = 1; i <= 10; i++) {
            if (i % 2 == 0) continue;
            sum += i;
        }
        System.out.println(sum);
        int n = 0;
        while (true) { n++; if (n == 5) break; }
        System.out.println(n);
        int k = 10;
        do { k -= 3; } while (k > 0);
        System.out.println(k);
        outer:
        for (int i = 0; i < 5; i++) {
            for (int j = 0; j < 5; j++) {
                if (i * j == 6) { System.out.println(i + "," + j); break outer; }
            }
        }
        String day = "SAT";
        switch (day) {
            case "SAT":
            case "SUN":
                System.out.println("weekend");
                break;
            default:
                System.out.println("weekday");
        }
        int x = 2;
        switch (x) {
            case 1: System.out.println("one");
            case 2: System.out.println("two");
            case 3: System.out.println("three");
        }
        String r = switch (x) {
            case 1 -> "unu";
            case 2 -> { String t = "do"; yield t + "i"; }
            default -> "multe";
        };
        System.out.println(r);
        System.out.println(x > 1 ? "mare" : "mic");
    }
}

abstract class Shape {
    protected String name;
    Shape(String name) { this.name = name; }
    abstract double area();
    String describe() { return name + " with area " + String.format("%.2f", area()); }
}
class Circle extends Shape {
    private double r;
    Circle(double r) { super("Circle"); this.r = r; }
    @Override double area() { return Math.PI * r * r; }
}
class Rectangle extends Shape {
    protected double w, h;
    Rectangle(double w, double h) { super("Rectangle"); this.w = w; this.h = h; }
    double area() { return w * h; }
}
class Square extends Rectangle {
    Square(double s) { super(s, s); name = "Square"; }
}
public class Main {
    public static void main(String[] args) {
        Shape[] shapes = { new Circle(1), new Rectangle(2, 3), new Square(2) };
        double total = 0;
        for (Shape s : shapes) {
            System.out.println(s.describe());
            total += s.area();
        }
        System.out.printf("Total: %.3f%n", total);
        System.out.println(shapes[2] instanceof Rectangle);
        System.out.println(shapes[0] instanceof Rectangle);
        Object o = shapes[1];
        if (o instanceof Rectangle r && r.w > 1) System.out.println("wide " + r.w);
    }
}

sealed interface Shape permits Circle, Rect {}
record Circle(double r) implements Shape {}
record Rect(double w, double h) implements Shape {
    Rect {
        if (w < 0 || h < 0) throw new IllegalArgumentException("negative");
    }
    double area() { return w * h; }
}
public class Main {
    static double area(Shape s) {
        return switch (s) {
            case Circle c -> Math.PI * c.r() * c.r();
            case Rect(double w, double h) -> w * h;
        };
    }
    static String size(Shape s) {
        return switch (s) {
            case Circle c when c.r() > 10 -> "big circle";
            case Circle c -> "circle";
            case Rect r -> "rect " + r.area();
        };
    }
    public static void main(String[] args) {
        Shape a = new Circle(1), b = new Rect(2, 3);
        System.out.printf("%.4f %.1f%n", area(a), area(b));
        System.out.println(size(new Circle(20)) + ", " + size(a) + ", " + size(b));
        System.out.println(b);
        System.out.println(new Rect(2, 3).equals(b) + " " + (new Rect(2, 3) == b));
        try { new Rect(-1, 2); } catch (IllegalArgumentException e) { System.out.println(e.getMessage()); }
        record Point(int x, int y) {}
        var p = new Point(3, 4);
        System.out.println(p.x() + p.y() + " " + p);
        Object o = p;
        if (o instanceof Point(int x, int y) && x < y) System.out.println("x<y");
    }
}

        Box(T value) { this.value = value; }
